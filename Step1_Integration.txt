============================================================
EKO SEMANTIC SEARCH -- STEP 1 (Resources) -- INTEGRATION GUIDE
============================================================

WHAT STEP 1 DOES
----------------
Adds a meaning-based (semantic) layer to the RESOURCES section of search
only. Keyword search (Resources_Search_New) runs unchanged; semantic
results are fused into the same Resources list via Reciprocal Rank
Fusion. Results that semantic found but keyword did NOT are flagged with
a small badge (toggleable). All other search sections (Pages, News,
Member Directory, Forum) are completely untouched.

Permissions: semantic ranking is restricted to the exact set of resource
IDs the user is allowed to see -- taken from Resources_Search_New's own
@mytable result set. No parallel permission logic.

--- 4c. replace the body of Search_FTS with the version below ---
The only new logic is the "if (_myClass.SemanticEnabled ...)" block and
the render loop honoring fused order + badge. getTable(), RemoveNoiseWords(),
and the keyword call are unchanged.

private int Search_FTS(string keywords)
{
    int total = 0;
    litContent.Text = "";
    litTitle.Text = "<h2>" + _myClass.Title + "</h2>";

    string comm = _myClass.StoredProcedure;
    string lcid = LCIDs[Convert.ToInt32(Session["Language"]) - 1];

    // Keyword path: noise-stripped term (as today)
    string searchTerm = RemoveNoiseWords(keywords, lcid);
    myFTS = new FullTextSearch.FullTextSearch(searchTerm);

    List<SqlParameter> param = new List<SqlParameter>();
    param.Add(new SqlParameter("@keywords", myFTS.NormalForm));
    param.Add(new SqlParameter("@lang", Convert.ToInt32(Session["Language"])));
    param.Add(new SqlParameter("@LCID", lcid));
    if ((comm == "SearchPagesFTS" || comm == "Resources_Search_New") && Session["LoggedInID"] != null)
        param.Add(new SqlParameter("@userid", Session["LoggedInID"].ToString()));

    // Fill the FULL dataset (Resources_Search_New returns @mytable too).
    DataSet ds = getDataSet(comm, param.ToArray());
    DataTable dt = ds.Tables[0];
    total = dt.Rows.Count;

    // ---- Semantic fusion: Resources section only ----
    bool doSemantic = _myClass.SemanticEnabled
                      && comm == "Resources_Search_New"
                      && Session["LoggedInID"] != null;

    if (doSemantic)
    {
        // keyword-ranked ids (dt is already ordered by score DESC)
        List<int> keywordIds = new List<int>();
        foreach (DataRow r in dt.Rows) keywordIds.Add(Convert.ToInt32(r["id"]));

        // permitted+filtered universe = @mytable, the LAST result set
        List<int> permitted = new List<int>();
        DataTable my = ds.Tables[ds.Tables.Count - 1]; // @mytable: single "id" column
        if (my != null && my.Columns.Contains("id"))
            foreach (DataRow r in my.Rows) permitted.Add(Convert.ToInt32(r["id"]));

        // ORIGINAL (un-stripped) term goes to the embedder
        List<SemanticHit> sem = ResourceSemanticSearch.Search(keywords, permitted, 50);

        if (sem.Count > 0)
        {
            List<FusedHit> fused = ResourceSemanticSearch.Fuse(keywordIds, sem, 50);

            // display rows: reuse keyword rows; fetch any semantic-only ids
            Dictionary<int, DataRow> rowById = new Dictionary<int, DataRow>();
            foreach (DataRow r in dt.Rows) rowById[Convert.ToInt32(r["id"])] = r;

            List<int> missing = new List<int>();
            foreach (FusedHit h in fused)
                if (!rowById.ContainsKey(h.ResourceId)) missing.Add(h.ResourceId);

            if (missing.Count > 0)
            {
                DataTable extra = getDataSet("Resources_GetByIds",
                    new SqlParameter[] { new SqlParameter("@Ids", string.Join(",", missing)) }
                ).Tables[0];
                foreach (DataRow r in extra.Rows) rowById[Convert.ToInt32(r["id"])] = r;
            }

            total = fused.Count;
            RenderHeader(total);

            int shown = _partial ? Math.Min(total, _records) : total;
            for (int i = 0; i < shown; i++)
            {
                FusedHit h = fused[i];
                DataRow r;
                if (!rowById.TryGetValue(h.ResourceId, out r)) continue;

                string badge = "";
                if (h.Source == HitSource.SemanticOnly && SearchConfig.ShowSemanticBadge)
                    badge = " <span class='sem-badge' title='Suggested by meaning-based search'>suggested</span>";

                litContent.Text += String.Format(
                    "<div><a href='{0}'>{1}</a>{2}</div><br>",
                    r["seo"].ToString(), r["title"].ToString(), badge);
            }

            TrackSearch(keywords, total);
            return total;
        }
        // sem.Count == 0 -> fall through to plain keyword rendering below
    }

    // ---- Original keyword-only rendering (unchanged behavior) ----
    if (dt.Rows.Count == 0) { litMessage.Text = "<h3>No results</h3>"; TrackSearch(keywords, 0); return 0; }
    RenderHeader(dt.Rows.Count);
    int n = _partial ? Math.Min(dt.Rows.Count, _records) : dt.Rows.Count;
    for (int i = 0; i < n; i++)
    {
        DataRow dr = dt.Rows[i];
        litContent.Text += String.Format("<div><a href='{0}'>{1}</a></div><br>",
            dr["seo"].ToString(), dr["title"].ToString());
    }
    TrackSearch(keywords, dt.Rows.Count);
    return total;
}

--- 4d. small helpers (add to the control) ---
// Same as the old getTable but returns the whole DataSet (so we can read @mytable)
private DataSet getDataSet(string cmd, SqlParameter[] param)
{
    SqlDataAdapter da = new SqlDataAdapter(cmd,
        new SqlConnection(ConfigurationManager.AppSettings.Get(_myClass.ConnectionString)));
    da.SelectCommand.CommandType = CommandType.StoredProcedure;
    da.SelectCommand.Parameters.AddRange(param);
    DataSet ds = new DataSet();
    da.Fill(ds);
    return ds;
}

private void RenderHeader(int totalRows)
{
    if (_partial && totalRows > _records)
        litMessage.Text = String.Format(
            "<h3>Displaying {0} of {1} {2} results - <a href='/{3}?{5}={4}'>View All</a></h3>",
            _records, totalRows, _myClass.Type, _myClass.AllResultsPage, SearchTerm, _myClass.QueryString_SearchParm);
    else
        litMessage.Text = String.Format("<h3>Displaying {0} {1} results</h3>", totalRows, _myClass.Type);
}

private void TrackSearch(string keywords, int count)
{
    using (SqlConnection conn = new SqlConnection(ConfigurationManager.AppSettings.Get("CMServer")))
    {
        SqlCommand cmd = new SqlCommand("SearchTracking", conn);
        cmd.CommandType = CommandType.StoredProcedure;
        cmd.Parameters.AddWithValue("@keywords", keywords);
        cmd.Parameters.AddWithValue("@ResultsCount", count);
        conn.Open(); cmd.ExecuteNonQuery();
    }
}

NOTE: DoSearch() currently calls Search_FTS then SearchTracking. Since
TrackSearch is now called inside Search_FTS, remove the SearchTracking
block from DoSearch to avoid double-logging (leave DoSearch's deepin++ and
the Search_FTS call).

============================================================
6. NOTES / SAFETY (step 1)
============================================================
- No generative AI here: every result is a real indexed resource row;
  nothing is invented. Semantic = better matching, not text generation.
- Permission safety: semantic candidates come only from @mytable, the
  same permission+category-filtered set the keyword proc trusts. Category,
  sub-category and Favourite filters therefore apply to both paths for free.
- Anonymous users: Resources are member-only; @userid is not passed and
  the section does not run for them -- unchanged from today.
- Graceful degradation: if the embedding API is unreachable, semantic
  returns empty and the section renders keyword-only (no user-visible error).
- The "standard"/FORMSOF keyword quirk and noise-word stripping affect
  only the keyword path; the semantic path embeds the raw query and is
  unaffected -- a visible quality win for awkward queries.
