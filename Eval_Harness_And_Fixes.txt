A) APOSTROPHE FIX  (SearchResults.ascx.cs)
============================================================
The existing RemoveNoiseWords() passes any term containing ' or "
through UNCHANGED, so an apostrophe reaches CONTAINS/FREETEXT and errors.
Add a sanitizer and apply it to the keyword term only. (The embedder
needs no escaping -- apostrophes are harmless there.)

--- add helper ---
private string SanitizeForFts(string term)
{
    if (string.IsNullOrEmpty(term)) return term;
    term = term.Replace('\u2018', '\'').Replace('\u2019', '\''); // curly -> straight
    term = term.Replace("'", "''");   // double for CONTAINS/FREETEXT
    return term;
}

--- apply where the keyword term is built (around line 160) ---

searchTerm = SanitizeForFts(searchTerm);          // <-- ADD

============================================================
C) LOGGING TABLE + PROC  (Resources DB, or wherever you prefer)
============================================================
Captures every query's results with score, source, and rank so you can
tune minScore from real data and show EKO what semantic adds.

CREATE TABLE dbo.SearchEval
(
    EvalId      INT IDENTITY PRIMARY KEY,
    RunUtc      DATETIME NOT NULL CONSTRAINT DF_SearchEval_RunUtc DEFAULT (GETUTCDATE()),
    Query       NVARCHAR(400) NOT NULL,
    UserId      INT NULL,
    ResourceId  INT NOT NULL,
    Source      VARCHAR(20) NOT NULL,   -- KeywordOnly | SemanticOnly | Both
    Score       REAL NULL,              -- cosine score if semantic, else NULL
    Rank        INT NOT NULL            -- 1-based position in fused list
);
GO

CREATE PROCEDURE dbo.SearchEval_Log
    @Query NVARCHAR(400), @UserId INT, @ResourceId INT,
    @Source VARCHAR(20), @Score REAL, @Rank INT
AS
BEGIN
    SET NOCOUNT ON;
    INSERT dbo.SearchEval (Query, UserId, ResourceId, Source, Score, Rank)
    VALUES (@Query, @UserId, @ResourceId, @Source, @Score, @Rank);
END
GO

--- C# logging call: only when ?debug=1 (so you don't log every public
    search). Add inside the Resources branch after 'fused' is built. ---
if (IsDebugViewer())
{
    string conn = SearchConfig.ConnString(SearchConfig.ResourcesConnName);
    int? uid = Session["LoggedInID"] != null
        ? (int?)Convert.ToInt32(Session["LoggedInID"]) : null;
    using (var cn = new System.Data.SqlClient.SqlConnection(conn))
    {
        cn.Open();
        for (int i = 0; i < fused.Count; i++)
        {
            FusedHit h = fused[i];
            float? sc = null;
            float tmp; if (semScore.TryGetValue(h.ResourceId, out tmp)) sc = tmp;
            using (var cmd = new System.Data.SqlClient.SqlCommand("dbo.SearchEval_Log", cn))
            {
                cmd.CommandType = System.Data.CommandType.StoredProcedure;
                cmd.Parameters.AddWithValue("@Query", keywords);
                cmd.Parameters.AddWithValue("@UserId", (object)uid ?? DBNull.Value);
                cmd.Parameters.AddWithValue("@ResourceId", h.ResourceId);
                cmd.Parameters.AddWithValue("@Source", h.Source.ToString());
                cmd.Parameters.AddWithValue("@Score", (object)sc ?? DBNull.Value);
                cmd.Parameters.AddWithValue("@Rank", i + 1);
                cmd.ExecuteNonQuery();
            }
        }
    }
}


============================================================
HOW TO TUNE minScore WITH THIS
============================================================
1. Set debug on: append &debug=1 to the member search URL.
2. Run a batch of queries: real ones + "dfgdetgerd" (gibberish) +
   known-item queries. Each logs to SearchEval.
3. Read the data:

   -- What scores does gibberish produce? (these define your noise floor)
   SELECT Query, ResourceId, Source, Score, Rank
   FROM dbo.SearchEval
   WHERE Query = 'dfgdetgerd'
   ORDER BY Rank;

   -- Score distribution of semantic hits across all test queries
   SELECT Source, COUNT(*) AS n,
          MIN(Score) AS lo, AVG(Score) AS avg, MAX(Score) AS hi
   FROM dbo.SearchEval
   WHERE Score IS NOT NULL
   GROUP BY Source;

4. Set Search.Embedding.MinScore just ABOVE the gibberish scores
   (likely ~0.30-0.40), recycle app pool, re-test:
     - gibberish should now return ~0 results
     - real mismatch queries should still return the good matches
5. Once tuned, you can turn logging off (it only runs under ?debug=1
   anyway) and flip Search.ShowSemanticBadge to false for members.
